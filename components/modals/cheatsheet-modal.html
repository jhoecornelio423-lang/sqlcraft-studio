Guía Rápida y Chuleta de Sintaxis SQL

1. Selección, Proyección y Alias (SELECT / AS / DISTINCT)

SELECT columna1 AS nuevo_nombre, columna2 FROM tabla;
SELECT (precio * 0.90) AS precio_oferta FROM videojuegos;
SELECT c.nombre, c.email FROM clientes AS c; -- Alias de tabla
SELECT DISTINCT genero FROM videojuegos;     -- Valores únicos

2. Filtrado de Filas (WHERE / Operadores)

WHERE precio > 50 AND stock > 0
WHERE pais IN ('España', 'México')
WHERE año_lanzamiento BETWEEN 2018 AND 2022
WHERE (genero = 'Rol' OR precio < 30) AND stock > 0

3. Búsqueda de Cadenas (LIKE)

WHERE titulo LIKE 'The%'      -- Empieza con 'The'
WHERE email LIKE '%@gmail.com' -- Termina con '@gmail.com'
WHERE apellido LIKE '_o%'      -- Segunda letra es 'o'

4. Ordenación y Paginación (ORDER BY / LIMIT / OFFSET)

ORDER BY precio DESC, titulo ASC
LIMIT 5                -- Obtener los primeros 5
LIMIT 5 OFFSET 10       -- Saltar 10 registros y tomar los siguientes 5

5. Funciones de Agregación (COUNT, SUM, AVG, ROUND)

SELECT COUNT(*) AS total_registros,
       ROUND(AVG(precio), 2) AS precio_promedio,
       SUM(stock) AS total_unidades,
       MIN(precio) AS menor_precio,
       MAX(precio) AS mayor_precio
FROM videojuegos;

6. Agrupación y Filtro de Grupos (GROUP BY / HAVING)

SELECT consola, COUNT(*) AS cantidad
FROM videojuegos
GROUP BY consola
HAVING COUNT(*) >= 3
ORDER BY cantidad DESC;

7. Combinación de Tablas (INNER JOIN / LEFT JOIN)

-- Solo coincidencias en ambas tablas
SELECT v.id, c.nombre, j.titulo
FROM ventas v
INNER JOIN clientes c ON v.cliente_id = c.id
INNER JOIN videojuegos j ON v.videojuego_id = j.id;

-- Todos los registros de la izquierda (incluso sin ventas)
SELECT j.titulo, v.id
FROM videojuegos j
LEFT JOIN ventas v ON j.id = v.videojuego_id
WHERE v.id IS NULL; -- Detectar los que no tienen venta

8. Subconsultas (Subqueries)

-- Subconsulta escalar
SELECT titulo, precio FROM videojuegos 
WHERE precio > (SELECT AVG(precio) FROM videojuegos);

-- Subconsulta con IN
SELECT nombre FROM clientes
WHERE id IN (SELECT cliente_id FROM ventas WHERE metodo_pago = 'Cripto');
